// A diagram that renders as the live Archify viewer, falling back to a still
// image.
//
// Set three attributes before including this partial:
//
//   :diagram-name: system-architecture
//   :diagram-alt: Kumoss system architecture: the edge, the core, the sidecars
//   :diagram-verb: trace a relationship
//   include::partial$interactive-diagram.adoc[]
//
// `diagram-name` is the shared base name of the two committed artifacts --
// `images/<name>.png` and `attachments/<name>.html` -- which `docs/diagrams/`
// produces together. See docs/diagrams/README.adoc.
//
// The page ships the still image and swaps the viewer in over it. The two
// render at the same apparent size for a given width -- the PNG is a 2048px
// capture of this same viewer, so scaling it down to the column gives back
// exactly the size the viewer would have drawn anyway -- so the still is a
// fallback, not a sharper alternative. What the viewer adds is the ability to
// zoom: at column width neither form is readable at rest (6-7px text), and
// only the live one lets a reader do something about it.

image::{diagram-name}.png["{diagram-alt}"]

[subs="attributes+"]
++++
<p class="archify-actions" data-archify-src="{attachmentsdir}/{diagram-name}.html" data-archify-title="{diagram-alt}">
  <span class="archify-hint">Pan, zoom, search, {diagram-verb}, switch themes.</span>
  <a class="archify-full" href="{attachmentsdir}/{diagram-name}.html" target="_blank" rel="noopener">Open in a new tab</a>
</p>
<script>
/* Progressive enhancement for the block above. Guarded so that a page holding
   several diagrams runs this once, and deferred to DOMContentLoaded because a
   parse-time script cannot yet see the diagrams below it. */
(function () {
  if (window.__archifyEmbed) return
  window.__archifyEmbed = true

  /* The viewer's own interactive chrome. Its toolbar and zoom bar are
     position:fixed, so they always reach the bottom of whatever viewport the
     iframe gives them -- measuring them would make the height measurement
     below a fixpoint that can never shrink. Same selector list as
     docs/diagrams/capture-png.mjs, which hides them instead; here they stay
     visible and are only excluded from the measurement. */
  var CHROME = '.toolbar, .diagram-nav, .focus-chip, .guided-views, .archify-toast, .no-print'

  var style = document.createElement('style')
  style.textContent = [
    '.archify-actions{display:flex;flex-wrap:wrap;gap:.35rem .9rem;align-items:baseline;',
    'margin-block:-.65rem 1.5rem;font-size:.875rem;line-height:1.5}',
    '.archify-hint{opacity:.72}',
    '.archify-frame{width:100%;border:0;display:block}',
    /* A viewer is ~800KB and takes real time to arrive, so it loads out of
       the way while the still stays on screen, and is only swapped in once
       it has something to show. Off-screen rather than display:none: a
       display:none iframe has no viewport, so the application inside would
       not lay out, and revealing it would show an empty frame anyway --
       which is the whole failure this avoids. */
    '.archify-frame[data-archify-loading]{position:absolute;left:-99999px;top:0}',
    /* Only while the viewer is live. The still image is centred by the theme
       inside the narrower .imageblock; the frame is centred by this margin
       inside the wider one. */
    '.imageblock.archify-live{margin-inline:auto}',
    /* The theme insets an image block's contents by 40px, which puts the
       diagram 40px right of every heading and paragraph on the page. Those
       do line up with each other: a heading's box starts 64px further left
       than its text, because the theme carves that gutter with a negative
       margin and an equal padding to hold the hover anchor. So the column
       edge is where the text sits, and the diagram was the one thing off it.
       Dropped on both forms, so activating a block does not shift it. */
    '.imageblock.archify-block > .content{padding-inline:0}',
  ].join('')
  document.head.appendChild(style)

  /* The .imageblock wrapper is what escapes the 820px .sectionbody cap
     (`width: min(calc(var(--content-max-width)*1.25), var(--doc-inset-cap))`),
     so the viewer is swapped in inside it rather than in place of it. */
  function blockFor (actions) {
    var el = actions.previousElementSibling
    while (el && !el.classList.contains('imageblock')) el = el.previousElementSibling
    return el
  }

  /* Width is the only lever on how large the viewer draws: its SVG comes out
     at roughly 0.9x the viewport width it is given, and viewport height barely
     moves it. But --doc-inset-cap holds .imageblock at 1025px however wide the
     window gets, which on a large display leaves the whole right half of the
     article column unused. Track the column instead, so a 2560px screen gets
     a 1490px frame rather than the still's 945px. */
  function columnWidth (block) {
    var column = block.closest('.doc')
    return column ? column.clientWidth : 0
  }

  function widen (block) {
    var width = columnWidth(block)
    if (width) block.style.width = width + 'px'
  }

  /* Height of the real content, which is not the same as the document height:
     the viewer is a full-viewport application, so both scrollHeight and body
     height just echo back the height we set. Measure the lowest visible
     content edge instead, ignoring the fixed chrome. */
  function contentHeight (frame) {
    var doc = frame.contentDocument
    if (!doc) return 0
    var view = doc.defaultView
    var root = doc.querySelector('.container') || doc.body
    var nodes = root.querySelectorAll('*')
    var bottom = 0
    for (var i = 0; i < nodes.length; i++) {
      var node = nodes[i]
      var computed = view.getComputedStyle(node)
      if (computed.display === 'none' || computed.visibility === 'hidden' || computed.opacity === '0') continue
      if (computed.position === 'fixed' || node.closest(CHROME)) continue
      var rect = node.getBoundingClientRect()
      if (!rect.width || !rect.height) continue
      bottom = Math.max(bottom, rect.bottom)
    }
    return bottom ? Math.ceil(bottom + 24) : 0
  }

  /* A wider frame makes the viewer draw a larger diagram, which makes it
     taller, which is the measurement that sized the frame. Settle it over a
     few bounded passes instead of assuming one is enough, and stop once the
     change is not worth another reflow.

     A measurement of zero means the viewer has not laid itself out yet, not
     that it never will: each one is a whole application, and end-to-end-flow
     loads five of them at once over a cold cache. Keep waiting, because
     giving up here strands the frame at its seeded height for good. */
  function fit (frame, pass, waits) {
    var height = contentHeight(frame)
    if (!height) {
      if ((waits || 0) < 60) setTimeout(function () { fit(frame, pass, (waits || 0) + 1) }, 250)
      return
    }
    var current = parseInt(frame.style.height, 10) || 0
    if (Math.abs(height - current) < 8) return
    frame.style.height = height + 'px'
    if ((pass || 0) < 3) setTimeout(function () { fit(frame, (pass || 0) + 1, 0) }, 250)
  }

  /* Replaces one still with its viewer, then calls `done` to free the slot for
     the next one -- on load, or on the breaker below, whichever comes first.

     The still stays on screen for the whole download. An earlier version put
     the frame in place the moment it was created, which left a blank grey
     panel where the diagram had been until ~800KB arrived: on a fast static
     server that is a flicker, on anything slower it is the whole reading
     experience. So the frame loads off-screen, at the size it is about to
     occupy, and the swap happens in one synchronous block. */
  function activate (actions, done) {
    var block = blockFor(actions)
    if (!block || block.classList.contains('archify-live')) return done()
    var content = block.firstElementChild || block
    var still = content.querySelector('img')

    var frame = document.createElement('iframe')
    frame.className = 'archify-frame'
    frame.title = actions.dataset.archifyTitle || 'Interactive diagram'
    /* Load it at the size it will be revealed at, so the viewer lays out for
       the real column and appearing costs no second reflow. Antora writes no
       width/height on the still, so one that has not decoded yet measures
       zero: fall back to a nominal height rather than a collapsed frame. */
    var seed = still ? Math.round(still.getBoundingClientRect().height) : 0
    frame.style.height = (seed > 120 ? seed : 620) + 'px'
    frame.style.width = (columnWidth(block) || block.clientWidth || 900) + 'px'
    frame.setAttribute('data-archify-loading', '')

    /* Everything that makes the frame visible, in one go, so no intermediate
       state is ever painted. The frame is never moved in the DOM --
       re-inserting an iframe reloads it, which would throw away the wait --
       so the still is removed around it instead. */
    function reveal () {
      frame.removeAttribute('data-archify-loading')
      frame.style.width = ''
      /* .imageblock is display:flex and .content is a shrink-wrapping flex
         item: the image sized it by its own intrinsic width, but an iframe
         has none, so .content would collapse to the UA's 300px iframe default
         and the frame's width:100% would resolve against an indefinite
         container. Pin .content to the flex container's width so the
         percentage has something definite to resolve against. */
      content.style.width = '100%'
      block.classList.add('archify-live')
      widen(block)
      for (var i = content.children.length - 1; i >= 0; i--) {
        if (content.children[i] !== frame) content.children[i].remove()
      }
    }

    /* One viewer loads at a time, so a frame that never fires load would hold
       the queue for good and strand every diagram below it as a still.
       Release the slot after a generous wait; this frame still reveals itself
       if it turns up late. */
    var settled = false
    function settle () {
      if (settled) return
      settled = true
      done()
    }
    setTimeout(settle, 15000)

    frame.addEventListener('load', function () {
      /* The viewer repeats its own title as an <h1> above the diagram, and
         every page that embeds one already carries that line as the section
         heading right above the frame. Two of the same sentence is one too
         many, and at column width the repeat wraps onto a second line and
         runs into the viewer's own toolbar. Hidden here rather than removed
         from the specification: a title is required there, and the "Open in a
         new tab" page has no surrounding heading to make it redundant. Same
         document, same origin, so this is a stylesheet and not a message. */
      var inner = frame.contentDocument
      if (inner) {
        var hide = inner.createElement('style')
        hide.textContent = '.container > .header{display:none}'
        inner.head.appendChild(hide)
      }

      /* `load` means the viewer's assets arrived and its scripts ran, not that
         it has drawn: the SVG appears in a later layout pass. Revealing on
         load alone shows the viewer's own empty panel for a frame or two, so
         wait for the drawing to exist -- briefly, and then reveal anyway,
         because a viewer that never draws must not keep the still forever. */
      var tries = 0
      ;(function ready () {
        var drawn = false
        try { drawn = !!(inner && inner.querySelector('svg')) } catch (e) { drawn = true }
        if (!drawn && ++tries <= 20) return void setTimeout(ready, 100)

        reveal()
        settle()

        /* Only now is the frame in the flow at its final width, so this is
           where the height measurement belongs. */
        setTimeout(function () { fit(frame, 0) }, 400)
        var width = 0
        new ResizeObserver(function () {
          /* Width only: observing our own height would loop against fit(). */
          var next = Math.round(frame.getBoundingClientRect().width)
          if (next === width) return
          width = next
          fit(frame, 0)
        }).observe(frame)
      })()
    })

    content.appendChild(frame)
    frame.src = actions.dataset.archifySrc
  }

  /* A page can hold five of these and each is a whole application of roughly
     800KB, so they are loaded strictly one at a time.

     Serial, not merely staggered. An earlier version started one every 900ms,
     which on a slow origin simply put all five in flight together: a browser
     opens about six connections per origin, five long downloads take nearly
     all of them, and the next click on the navigation sidebar gets no socket
     at all. That is not a slow page, it is a page that has stopped answering
     -- the network panel shows the navigation cancelled with zero bytes
     transferred. One viewer at a time bounds the cost to a single connection
     however slow the origin is, and the reader keeps a still until its
     replacement is ready, so nothing is lost by waiting.

     Proximity decides the order and never whether. Two earlier versions let it
     decide whether, and both stranded the lower diagrams as stills for good:
     the reader saw a tiny illegible image where a viewer should be. The cause
     outlives either mechanism. Antora writes no width or height on a still, so
     every image reflows everything below it as it decodes -- which means a
     block can be carried into range with no event to announce it. An
     IntersectionObserver's one-shot entry arrives against a position the block
     no longer holds; a rect read on scroll is never stale, but nothing reads it
     once the reader stops scrolling. Working through the whole queue removes
     that class of failure instead of widening the window it hides in: being
     near a diagram only brings it forward. */
  function enhance () {
    var pending = []
    var actions = document.querySelectorAll('.archify-actions[data-archify-src]')
    for (var i = 0; i < actions.length; i++) {
      var block = blockFor(actions[i])
      if (!block) continue
      /* Marks the blocks this script owns, so the stylesheet above can reach
         them while they are still stills. */
      block.classList.add('archify-block')
      pending.push(actions[i])
    }
    if (!pending.length) return

    var queued = false
    var running = false

    function stop () {
      window.removeEventListener('scroll', schedule)
      window.removeEventListener('resize', schedule)
    }

    /* Leaving the page ends the queue. Without this, a click on the navigation
       sidebar races a pump that is about to start another ~800KB viewer: the
       browser asks for the next document while this page opens one more
       connection for a viewer nobody will ever see, and on a slow origin the
       navigation is what waits. The reader reads that as a page that has
       stopped answering. Nothing below needs the queue once the document is
       on its way out. */
    window.addEventListener('pagehide', function () {
      pending.length = 0
      stop()
    })

    /* One observer for the whole column, not one per diagram. Every live block
       tracks the same element, so five separate observers on it did the same
       work five times over on every resize. */
    var column = document.querySelector('.doc')
    if (column) {
      new ResizeObserver(function () {
        var live = document.querySelectorAll('.imageblock.archify-live')
        for (var i = 0; i < live.length; i++) widen(live[i])
      }).observe(column)
    }

    /* The one place a viewer is ever started. Nothing else calls activate, so
       there is exactly one download in flight no matter what triggers a
       pump -- scroll, resize, or the previous one finishing. */
    function pump () {
      if (running || !pending.length) return
      running = true
      activate(pending.shift(), function () {
        running = false
        if (pending.length) setTimeout(pump, 250)
        else stop()
      })
    }

    /* Reorders what is left so anything the reader can nearly see goes next,
       and never activates anything itself. A diagram already passed keeps its
       place in the queue: the saving worth having is not loading five viewers
       at once, not reclaiming one that has scrolled off the top. */
    function promote () {
      queued = false
      var visible = []
      var rest = []
      for (var i = 0; i < pending.length; i++) {
        var block = blockFor(pending[i])
        var close = block && block.getBoundingClientRect().top <= window.innerHeight + 400
        ;(close ? visible : rest).push(pending[i])
      }
      pending = visible.concat(rest)
      pump()
    }

    function schedule () {
      if (queued) return
      queued = true
      requestAnimationFrame(promote)
    }

    window.addEventListener('scroll', schedule, { passive: true })
    window.addEventListener('resize', schedule)
    promote()
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', enhance)
  else enhance()
})()
</script>
